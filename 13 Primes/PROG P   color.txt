PROG P                                               
BARC 2         
BARH 30
2
BARC 3 
3
I=4
UPTO=23
WHIL I LE UPTO                         
DO      
   PRIM=1
   IF I MOD 2 EQ 0
   THEN PRIM=0
   ELSE  
      J=3    
      WHIL J LE (I * .5)
       DO 
          IF I MOD J EQ 0
          THEN PRIM=0
             EXIT
          ELSE J=J+2 
          ENDI
       ENDW
   ENDI 
   BARC 1+(I MOD 7)
   IF PRIM EQ 1                                                     
   THEN BARH 30
      I
   ELSE
      BARH -30
   ENDI
   I=I+1
ENDW 
ENDP
