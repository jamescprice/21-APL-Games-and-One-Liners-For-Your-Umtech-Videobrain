prog mastermind
z= rand 5 5 5
"guess the 3 numbers"
i=99 99 99
whil 1 
do                  
   i=keyb                 
   if i(1) eq 99       
   then exit
   endi
   if 3 eq +/(z eq i) 
   then "you win" 
      exit
   else
      c= +/ (z eq i)

      j=1
      r=0
      whil j le 3    //correct color wrong spot
      do
         if z(j) ne i(j)
         then
            r=r+ (+/(z(j) eq i ) )
         endi
      j=j+1  
      endw
   "white/red"
   c,r
   endi
endw
