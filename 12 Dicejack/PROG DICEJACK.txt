PROG DICEJACK   
Z=RAND 6 6
Z
B=0
H=1
WHIL H EQ 1
DO
   ‘HIT? 1/0’ 
   C=KEYB
   IF C EQ 0 
   THEN H = 0
   ENDI
   IF C EQ 1
   THEN 
      Z=Z,RAND 6
      Z
      IF +/Z GT 21
      THEN "BUSTED"
         B=1
         EXIT
      ENDI
   ENDI
ENDW
IF B NE 1
THEN 
   D=0   //DEALER NOT BUSTED
   X=RAND 6 6
   X
   WHIL +/Z GE +/X    
   DO
      X=X,RAND 6
      X
      IF +/X GT 21
      THEN "I BUSTED"
         D=1
         EXIT
      ENDI    
   ENDW
   IF D EQ 1
   THEN "YOU WIN"   
   ELSE 
     IF +/X GT +/Z THEN "I WIN"
        ELSE "YOU WIN" 
     ENDI  
   ENDI
ENDI

ENDP
